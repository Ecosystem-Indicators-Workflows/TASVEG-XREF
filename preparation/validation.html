library(readr)

here::i_am("preparation/validation.qmd")

6.1 Read membership table

Read data directly from the data frame and coerce it to a membership_tbl object:

TASVEG_memb <- read_membership_tbl(
    TASVEG_xref,
    source_id = "VEGCODE",
    target_id = "EFG",
    membership = "memb",
    interpreter = "expert",
    round = "round"
)

    TASVEG_xref,
    TASVEG_clean,
    file = here::here(output_dir, "memb-tables.rda")
)
